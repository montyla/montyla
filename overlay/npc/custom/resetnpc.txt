//===== rAthena Script =======================================
//= Reset NPC
//===== Description: =========================================
//= Resets skills, stats, or both.
//===== Additional Comments: =================================
//= 1.0 First Version
//= 1.1 Optimized for the greater good. [Kisuka]
//= 1.2 Cleaning [Euphy]
//= 1.3 All statuses removed upon skill reset. [Euphy]
//= 1.4 Compressed Script, Added limit use option [Stolao]
//=	Changed set -> setarray, Improved text with F_InsertPlural
//= 1.5 Added sc_end_class to reset related status changes [sader1992]
//============================================================
prontera,150,193,4	script	Reset de Status	124,{
	//	 		Skills,	Stats,	Both,	Limit
	setarray .@Reset,	5000,	5000,	9000,	0;
	mes "[Reset de Status]";
	if(.@Reset[3] && reset_limit > .@Reset[3]) {
		mes "Desculpe, você só pode resetar "+.@Reset[3]+((.@Reset[3]>1)?" vezes":" vez")+" na vida.";
		close;
	}
	mes "Eu reseto seus status e habilidades.";
	mes "Resetar atributos: "+ callfunc("F_InsertComma",.@Reset[1]) +"z";
	mes "Resetar habilidades: "+ callfunc("F_InsertComma",.@Reset[0]) +"z";
	mes "Resetar os dois: "+ callfunc("F_InsertComma",.@Reset[2]) +"z";
	if(.@Reset[3]) mes "Você só pode resetar "+.@Reset[3]+((.@Reset[3]>1)?" vezes":" vez")+", então use com cuidado.";
	mes "Qual serviço você deseja?";
	next;
	set .@i,(select("^FF3355Resetar habilidades:Resetar atributos:Resetar os dois^000000:Cancelar"));
	if(.@i > 3) close;
	mes "[Reset de Status]";
	if (Zeny < .@Reset[.@i-1]) {
		mes "Desculpe, você não tem Zeny suficiente.";
		close;
	}
	if(.@Reset[3]){
		mes "Você só pode resetar "+.@Reset[3]+((.@Reset[3]>1)?" vezes":" vez")+" na vida. Tem certeza?";
		if(select("Deixa eu pensar:Pode resetar") == 1) close;
	}
	set Zeny, Zeny-.@Reset[.@i-1];
	if(.@i&1){
		sc_end_class;
		ResetSkill;
	}
	if(.@i&2) ResetStatus;
	mes "Pronto! Tudo resetado.";
	if(.@Reset[3]) set reset_limit,reset_limit + 1;
	close;
}
