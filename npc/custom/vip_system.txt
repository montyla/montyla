-	script	VIP_TIME	-1,{
OnPCLoginEvent:
	if (vip_status(VIP_STATUS_ACTIVE)) {
		.@remain = vip_status(VIP_STATUS_REMAINING); // seconds

		dispbottom "Remaining VIP duration: " + Time2Str(.@remain + gettimetick(2));
		sc_start SC_VIPSTATE, .@remain * 1000, 1;
	}
	end;

OnPCLogoutEvent:
	if (vip_status(VIP_STATUS_ACTIVE)) {
		sc_end SC_VIPSTATE; // remove the buff to update the timer on login
	}
	end;
}

-	script	VIP_STORAGE	-1,{
OnInit:
	// Keep the default @storage disabled for normal players/groups.
	// This scripted command opens storage only when the account has active VIP.
	bindatcmd "storage", strnpcinfo(3) + "::OnStorage", 0, 99;
	end;

OnStorage:
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		dispbottom "O comando @storage esta disponivel apenas para jogadores VIP.";
		end;
	}

	openstorage;
	end;
}
